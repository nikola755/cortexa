~/cortexa

quick install

# clone and install
$ git clone https://github.com/nikola755/cortexa.git ~/cortexa
$ cd ~/cortexa
$ chmod +x install.sh
$ ./install.sh

options

$ ./install.sh --fast           # skip all prompts
$ ./install.sh --skip-packages  # only restore configs
$ ./install.sh --skip-backup    # don't backup existing
